create sequence if not exists public.student_number_seq start 1; create or replace function public.generate_student_id() returns text language plpgsql as $$ begin return 'ITI' || lpad(nextval('public.student_number_seq')::text, 4, '0'); end; $$; create or replace function public.handle_new_user() returns trigger language plpgsql security definer set search_path = public as $$ begin insert into public.profiles (id, student_id, full_name, phone, course) values ( new.id, public.generate_student_id(), coalesce(new.raw_user_meta_data->>'full_name', ''), new.raw_user_meta_data->>'phone', new.raw_user_meta_data->>'course' ); return new; end; $$; create trigger on_auth_user_created after insert on auth.users for each row execute procedure public.handle_new_user();
